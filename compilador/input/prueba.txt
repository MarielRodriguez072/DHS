int  escaneo (int a, int b, int c, int f) {
    int r = 0;

    if (a > b && b > c) {
        r = 1;
    }

    f = 9;

    while (r > 0 && (a > 0 && b > 0)) {
        r = r - 1;
    }

    return r;
}

int main() {
    int x = 5;
    int y = 4;
    int z = 3;
    int resultado;
    int f = 67;

    resultado = escaneo(x, y, z, f);

    if (resultado >= 0 && resultado <= 10) {
        resultado = resultado + 1;
    }

    return 0;
}
